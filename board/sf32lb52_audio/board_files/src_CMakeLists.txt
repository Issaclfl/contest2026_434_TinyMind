set(SRCS sifli_ap.c sifli_gpio.c sf32lb52_buttons.c bsp_init.c bsp_lcd_tp.c bsp_pinmux.c bsp_power.c bsp_audio.c bsp_audio_hw.c)

nuttx_add_romfs(
  NAME
  etc
  MOUNTPOINT
  etc
  RCSRCS
  etc/init.d/rcS
  etc/init.d/rc.sysinit
  RCRAWS
  etc/group
  etc/1.txt
  etc/clips/light_on.pcm
  etc/clips/light_off.pcm
  etc/clips/red_on.pcm
  etc/clips/blue_on.pcm
  etc/clips/offline.pcm
  etc/clips/ok.pcm
  etc/clips/unknown.pcm
  etc/clips/temp_high.pcm
  etc/clips/lamp_on.pcm
  etc/clips/lamp_off.pcm
  etc/clips/ac_on.pcm
  etc/clips/ac_off.pcm
  etc/clips/bright_set.pcm
  etc/clips/ct_set.pcm
  etc/clips/no_reply.pcm
  etc/clips/no_device.pcm
  etc/clips/green_on.pcm
  etc/clips/white_on.pcm
  etc/clips/yellow_on.pcm
  etc/clips/rgb_on.pcm
  etc/clips/rgb_off.pcm
  etc/clips/led_blink.pcm
  etc/clips/led_breath.pcm
  etc/fonts/NotoSansSC-sub.ttf   # tools/subset_font.py 生成
  PATH
  ${CMAKE_CURRENT_LIST_DIR}/etc)

target_link_libraries(board PRIVATE romfs_etc)

target_sources(board PRIVATE ${SRCS})

target_compile_definitions(board PRIVATE SOC_BF0_HCPU
                                         SF32LB52X
                                         HAL_TICK_PER_SECOND=1000
                                         USE_HAL_DRIVER)

target_include_directories(board PRIVATE ${NUTTX_CHIP_ABS_DIR}/include)
target_include_directories(board PRIVATE ${NUTTX_CHIP_ABS_DIR}/../drivers/Include)
target_include_directories(board PRIVATE ${NUTTX_CHIP_ABS_DIR}/../drivers/cmsis/Include)
target_include_directories(board PRIVATE ${NUTTX_CHIP_ABS_DIR}/../drivers/cmsis/sf32lb52x)
target_include_directories(board PRIVATE ${NUTTX_CHIP_ABS_DIR}/../external/CMSIS/Include)
target_include_directories(board PRIVATE ${NUTTX_CHIP_ABS_DIR}/../boards/include/config/sf32lb52x)

target_compile_options(board PRIVATE
    $<$<COMPILE_LANGUAGE:C>:-w>
    $<$<COMPILE_LANGUAGE:CXX>:-w>)

if(CONFIG_CRYPTO)
  nuttx_add_application(
    NAME
    md5_test
    STACKSIZE
    4096
    PRIORITY
    100
    SRCS
    md5_test.c)
endif()
